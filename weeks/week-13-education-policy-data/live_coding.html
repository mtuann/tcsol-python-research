plt.rcParams.update({"font.size": 11, "axes.titlesize": 14, "axes.labelsize": 11})
plot_timeline = timeline.drop_duplicates("doc_id").sort_values("issue_date_parsed")

ax.scatter(plot_timeline["issue_date_parsed"], range(len(plot_timeline)), color="#2f63ea", s=90)
for y, (_, row) in enumerate(plot_timeline.iterrows()):
    ax.text(row["issue_date_parsed"], y + 0.12, f"{row['doc_id']} · {row['source_type']}", fontsize=9, ha="center")

top_area = policy_area_summary.iloc[0]

    f"In the synthetic Week 13 policy-source coding dataset, {top_area['policy_area']} is one of the most frequent policy areas, "
    f"with {int(top_area['row_count'])} of {len(policy)} coded rows ({top_area['percent']}%). Several other areas have the same small count, so this should be read as a sample pattern rather than a ranking. The table also includes policy plans, "
